	generic = no
	ai = yes
	potential = { tag = SER }

eu4_2k_serbia_identity_missions = {
	slot = 5
	generic = no
	ai = yes
	potential = { tag = SER }
	has_country_shield = yes

	eu4_2k_ser_resolve_federal_legacy = {
		icon = mission_monarch_in_throne_room
		position = 1
		trigger = {
			OR = {
				has_country_flag = eu4_2k_serbia_formed_by_mission
				AND = { NOT = { owns = 138 } NOT = { owns = 4754 } }
				AND = { owns = 138 owns = 4754 }
			}
		}
		effect = { add_country_modifier = { name = eu4_2k_ser_resolved_federal_legacy_mod duration = 3650 } add_prestige = 10 }
	}

	eu4_2k_ser_define_kosovo_policy = {
		icon = mission_forts
		position = 2
		required_missions = { eu4_2k_ser_resolve_federal_legacy }
		trigger = {
			OR = {
				has_country_flag = eu4_2k_kosovo_settlement_reached
				AND = { owns = 1766 1766 = { NOT = { unrest = 3 } } }
				AND = { NOT = { owns = 1766 } exists = KOS KOS = { has_opinion = { who = ROOT value = 0 } } }
			}
		}
		effect = { add_country_modifier = { name = eu4_2k_ser_kosovo_policy_mod duration = 5475 } add_dip_power = 50 }
	}

	eu4_2k_ser_invest_in_belgrade = {
		icon = mission_early_game_buildings
		position = 3
		required_missions = { eu4_2k_ser_define_kosovo_policy }
		trigger = { owns = 4239 4239 = { development = 35 } }
		effect = { 4239 = { add_base_tax = 1 add_base_production = 1 add_base_manpower = 1 } add_prestige = 10 }
	}

	eu4_2k_ser_modern_serbia = {
